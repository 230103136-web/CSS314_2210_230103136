from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable


@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float
    max_ms: float


def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037
    for b in s.encode("utf-8"):
        h ^= b
        h = (h * 1099511628211) & 0xFFFFFFFFFFFFFFFF
    return h


def run_benchmark(
    fn: Callable[[], None], warmup_runs: int = 3, timed_runs: int = 9
) -> BenchStats:
    for _ in range(warmup_runs):
        fn()
    samples = []
    for _ in range(timed_runs):
        t0 = time.perf_counter_ns()
        fn()
        t1 = time.perf_counter_ns()
        samples.append((t1 - t0) / 1e6)
    samples.sort()
    mid = timed_runs // 2
    med = (
        samples[mid]
        if timed_runs % 2 == 1
        else 0.5 * (samples[mid - 1] + samples[mid])
    )
    q25 = samples[timed_runs // 4]
    q75 = samples[(3 * timed_runs) // 4]
    return BenchStats(samples[0], med, q75 - q25, samples[-1])


def emit_verify_tag(
    task_id: str,
    student_seed: int,
    primary_metric: float,
    secondary_metric: float,
) -> str:
    hw = mp.cpu_count()
    payload = (
        f"{task_id}|{student_seed}|{hw}|"
        f"{primary_metric:.4f}|{secondary_metric:.4f}"
    )
    sig = fnv1a_hash(payload)
    return f"VERIFY_HASH[{task_id}]: {sig:016X} (P_log={hw})"

def fs_worker(shm_name: str, slot_idx: int, iters: int) -> None:
    shm = shared_memory.SharedMemory(name=shm_name)
    arr = (ctypes.c_uint64 * 512).from_buffer(shm.buf)
    acc = arr[slot_idx]
    mask = 0xFFFFFFFFFFFFFFFF
    mult = 6364136223846793005
    for i in range(iters):
        acc = (acc * mult + (i | 1)) & mask
        arr[slot_idx] = acc
    del arr
    shm.close()


def run_fs_trial(shm_name: str, p: int, stride: int, total_ops: int) -> None:
    iters = total_ops // p
    procs = [
        mp.Process(target=fs_worker, args=(shm_name, t * stride, iters))
        for t in range(p)
    ]
    for proc in procs:
        proc.start()
    for proc in procs:
        proc.join()

def gemm_v1_worker(a_name: str, b_name: str, c_name: str, i_start: int, i_end: int) -> None:
    sa, sb, sc = shared_memory.SharedMemory(name=a_name), shared_memory.SharedMemory(name=b_name), shared_memory.SharedMemory(name=c_name)
    A, B, C = (ctypes.c_double * TOTAL_ELEMS).from_buffer(sa.buf), (ctypes.c_double * TOTAL_ELEMS).from_buffer(sb.buf), (ctypes.c_double * TOTAL_ELEMS).from_buffer(sc.buf)
    for i in range(i_start, i_end):
        row_i = i * N
        for j in range(N):
            dot = 0.0
            for k in range(N):
                dot += A[row_i + k] * B[k * N + j]
            C[row_i + j] = dot
    del A, B, C
    sa.close(); sb.close(); sc.close()

import math
import multiprocessing as mp
import time
from harness import emit_verify_tag, run_benchmark

N_ITEMS = 12_000


def compute_item(i: int) -> float:
    k = 1 + (i // 60) ** 2
    acc = (i + 1) * 0.001
    for _ in range(k):
        acc = math.sin(acc) + math.cos(acc) + math.sqrt(abs(acc) + 1.0)
    return acc


def static_block_worker(tid: int, p: int, worker_times) -> None:
    t0 = time.perf_counter_ns()
    chunk = N_ITEMS // p
    start = tid * chunk
    end = N_ITEMS if tid == p - 1 else (tid + 1) * chunk
    for i in range(start, end):
        compute_item(i)
    worker_times[tid] = (time.perf_counter_ns() - t0) / 1e6


def static_cyclic_worker(tid: int, p: int, worker_times) -> None:
    t0 = time.perf_counter_ns()
    for i in range(tid, N_ITEMS, p):
        compute_item(i)
    worker_times[tid] = (time.perf_counter_ns() - t0) / 1e6


def dynamic_chunk_worker(tid: int, chunk_size: int, counter, lock, worker_times) -> None:
    t0 = time.perf_counter_ns()
    while True:
        with lock:
            start = counter.value
            if start >= N_ITEMS:
                break
            counter.value = start + chunk_size
        end = min(start + chunk_size, N_ITEMS)
        for i in range(start, end):
            compute_item(i)
    worker_times[tid] = (time.perf_counter_ns() - t0) / 1e6


def evaluate_strategy(name: str, p: int, worker_times, runner_fn) -> tuple[float, float]:
    imbalances = []

    def wrapped():
        runner_fn()
        times = [worker_times[t] for t in range(p)]
        mean_t = sum(times) / p
        imbalances.append(max(times) / mean_t if mean_t > 0 else 1.0)

    stats = run_benchmark(wrapped, warmup_runs=1, timed_runs=5)
    med_lambda = sorted(imbalances)[len(imbalances) // 2]
    print(f"{name:<32} | {stats.median_ms:>11.2f} | {stats.iqr_ms:>9.2f} | {med_lambda:>10.4f}")
    return stats.median_ms, med_lambda
