The blocked outputs were compared with the reordered reference with a max-absolute-error threshold of 1e-9; the script raises an exception on failure. The compute worker performs 16 multiply-adds per inner iteration, counted as 32 FLOPs, so n//32 iterations correspond to n nominal FLOPs per worker.

## Method and compliance notes

- All numerical tables and VERIFY_HASH lines below are retained verbatim from the original local run; no new measurements or hashes were generated during this editorial revision.
- The embedded scripts are the **original measured implementation**, retained for reproducibility. Changing the implementation requires a new local run and regenerated hashes.

### harness.py
```python
from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable

@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float

def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037
    for b in s.encode('utf-8'):
        h = ((h ^ b) * 1099511628211) & 0xFFFFFFFFFFFFFFFF
    return h

def run_benchmark(fn: Callable[[], None], warmup: int = 2, runs: int = 7) -> BenchStats:
    for _ in range(warmup): fn()
    samples = []
    for _ in range(runs):
        t0 = time.perf_counter_ns()
        fn()
        samples.append((time.perf_counter_ns()-t0)/1e6)
    samples.sort()
    n = len(samples)
    return BenchStats(samples[0], samples[n//2], samples[(3*n)//4]-samples[n//4])

def emit_verify_tag(phase: str, seed: int, m1: float, m2: float) -> str:
    hw = mp.cpu_count()
    sig = fnv1a_hash(f'{phase}|{seed}|{hw}|{m1:.4f}|{m2:.4f}')
    return f'VERIFY_HASH[{phase}]: {sig:016X} (P_log={hw})'
```

```python
import ctypes,math,multiprocessing as mp,time
from multiprocessing import shared_memory
from harness import run_benchmark,emit_verify_tag
ID=230103136;N=256

def copy_worker(args):
    a,b,s,e=args
    x=shared_memory.SharedMemory(name=a);y=shared_memory.SharedMemory(name=b)
    x.buf[s:e]=y.buf[s:e]
    x.close();y.close()

def fma_worker(n):
    v=[0.01*(i+1) for i in range(16)]
    for _ in range(n//32):
        for j in range(16):v[j]=v[j]*1.00000001+0.00000001
    return sum(v)

def gemm_worker(args):
    an,bn,cn,start,end,variant,bs=args

    A=a.buf.cast('d');B=b.buf.cast('d');C=c.buf.cast('d');n=N
    if variant==1:
```

### task4_scheduling.py
```python
import math,multiprocessing as mp,time
from harness import emit_verify_tag
ID=230103136;N=8000

def work(i):
    k=1+(i//40)**2
    x=(i+1)*0.0001
    for _ in range(k):x=math.sin(x)+math.cos(x)*0.1+math.sqrt(abs(x)+0.001)*0.01
    return x

def worker(rank,p,mode,chunk,counter,times):
    t=time.perf_counter()
    if mode=='block':
        for i in range(N*rank//p,N*(rank+1)//p):work(i)
    elif mode=='cyclic':
        for i in range(rank,N,p):work(i)
    else:
        while True:
            with counter.get_lock():
                s=counter.value;counter.value+=chunk
            if s>=N:break
            for i in range(s,min(s+chunk,N)):work(i)
    times[rank]=time.perf_counter()-t

def trial(p,mode,chunk):
    counter=mp.Value('i',0);times=mp.Array('d',p)
    processes=[mp.Process(target=worker,args=(j,p,mode,chunk,counter,times)) for j in range(p)]
    t=time.perf_counter()
    for proc in processes:proc.start()
    for proc in processes:proc.join()
    if any(proc.exitcode for proc in processes):raise RuntimeError('Worker failed')
    wall=time.perf_counter()-t
    return wall,max(times)/(sum(times)/p)

def run(p):
    rows=[]
    for mode,chunk in [('block',0),('cyclic',0)]+[('dynamic',x) for x in (1,8,64,512)]:
        results=[trial(p,mode,chunk) for _ in range(3)]
        results.sort()
        wall,imb=results[1]
        rows.append((mode,chunk,wall,imb))
    best=min((r for r in rows if r[0]=='dynamic'),key=lambda x:x[2])
    return rows,best,emit_verify_tag('PHASE4',ID,rows[0][2],best[2])
if __name__=='__main__':print(run(4))
```
